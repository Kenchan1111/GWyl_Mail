```python
from flask import Flask, request, jsonify
import tempfile
import subprocess
import json
from pathlib import Path

app = Flask(__name__)

@app.route('/create-proof', methods=['POST'])
def create_proof_api():
    email_data = request.json
    
    # Sauvegarde temporaire email
    with tempfile.NamedTemporaryFile(mode='w', suffix='.eml', delete=False) as f:
        f.write(build_eml_file(email_data))
        eml_path = f.name
    
    try:
        # Appel CLI GWyl
        cmd = [
            'gwyl-mail', 'create-proof',
            eml_path,
            '--identity', email_data['from']['emailAddress'],
            '--out', '/tmp/proof.json'
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode == 0:
            proof = json.loads(Path('/tmp/proof.json').read_text())
            return jsonify(proof)
        else:
            return jsonify({'error': result.stderr}), 500
            
    finally:
        Path(eml_path).unlink(missing_ok=True)

def build_eml_file(email_data):
    # Construction fichier EML standard
    return f"""From: {email_data['from']['emailAddress']}
To: {email_data['to'][0]['emailAddress']}
Subject: {email_data['subject']}

{email_data['body']}
"""

if __name__ == '__main__':
    app.run(port=17362)  # Port dédié
```

```javascript
    return await createProofForCurrentEmail();
  }
  if (message.action === "getProofStatus") {
    return await getProofStatus();
  }

async function createProofForCurrentEmail() {
  try {
    // Récupérer les détails de l'email en cours
    const composeDetails = await messenger.compose.getComposeDetails(tab.id);
    
    const emailData = {
      from: composeDetails.from,
      to: composeDetails.to,
      cc: composeDetails.cc,
      bcc: composeDetails.bcc,
      subject: composeDetails.subject,
      body: composeDetails.body,
      attachments: await getAttachmentInfo(composeDetails.attachments)
    };

    // Appeler le service local GWyl
    const proof = await callGWylDaemon(emailData);
    
    // Ajouter la preuve en pièce jointe
    await attachProofToEmail(proof, tab.id);
    
    return { success: true, proofId: proof.message_id };
    
  } catch (error) {
    console.error("GWyl proof creation failed:", error);
    return { success: false, error: error.message };
  }
}

async function callGWylDaemon(emailData) {
  // Appel au service local sur le port dédié
  const response = await fetch('http://localhost:17362/create-proof', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(emailData)
  });
  
  if (!response.ok) {
    throw new Error(`GWyl service error: ${response.statusText}`);
  }
  
  return await response.json();
}

async function attachProofToEmail(proof, tabId) {
  const proofBlob = new Blob(
    [JSON.stringify(proof, null, 2)], 
    { type: 'application/json' }
  );
  
  await messenger.compose.addAttachment(tabId, {
    file: proofBlob,
    name: `email-proof-${proof.message_id}.json`
  });
}

async function getAttachmentInfo(attachments) {
  const attachmentInfo = [];
  for (const attachment of attachments) {
    attachmentInfo.push({
      name: attachment.name,
      size: attachment.size,
      type: attachment.contentType
    });
  }
  return attachmentInfo;
}
```

### **5. Service Local Amélioré**
```python
# gwyl-daemon.py
from flask import Flask, request, jsonify, send_file
import tempfile
import subprocess
import json
import os
from pathlib import Path
from threading import Lock
import logging

app = Flask(__name__)
logging.basicConfig(level=logging.INFO)
proof_lock = Lock()

@app.route('/create-proof', methods=['POST'])
def create_proof_api():
    with proof_lock:  # Évite les races conditions
        return _create_proof_impl(request)

def _create_proof_impl(request):
    try:
        email_data = request.json
        logger.info(f"Creating proof for email from {email_data.get('from', 'unknown')}")
        
        # Création fichier EML temporaire
        with tempfile.NamedTemporaryFile(mode='w', suffix='.eml', delete=False) as f:
            eml_content = build_eml_content(email_data)
            f.write(eml_content)
            eml_path = f.name
        
        proof_path = Path(tempfile.mktemp(suffix='.json'))
        
        try:
            # Appel CLI GWyl
            identity = extract_identity(email_data['from'])
            
            cmd = [
                'python', '-m', 'gwyl_mail.cli', 'create-proof',
                eml_path,
                '--identity', identity,
                '--out', str(proof_path)
            ]
            
            logger.info(f"Executing: {' '.join(cmd)}")
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=60)
            
            if result.returncode == 0 and proof_path.exists():
                proof_data = json.loads(proof_path.read_text())
                logger.info(f"Proof created successfully: {proof_data['message_id']}")
                return jsonify(proof_data)
            else:
                logger.error(f"GWyl CLI failed: {result.stderr}")
                return jsonify({
                    'error': 'Proof creation failed',
                    'details': result.stderr
                }), 500
                
        finally:
            # Nettoyage
            Path(eml_path).unlink(missing_ok=True)
            proof_path.unlink(missing_ok=True)
            
    except Exception as e:
        logger.error(f"Unexpected error: {str(e)}")
        return jsonify({'error': str(e)}), 500

def build_eml_content(email_data):
    """Convertit les données email en format EML standard"""
    lines = []
    
    # Headers
    lines.append(f"From: {email_data['from']}")
    lines.append(f"To: {', '.join([to['email'] for to in email_data.get('to', [])])}")
    
    if email_data.get('cc'):
        lines.append(f"Cc: {', '.join([cc['email'] for cc in email_data['cc']])}")
    
    lines.append(f"Subject: {email_data.get('subject', '')}")
    lines.append("MIME-Version: 1.0")
    lines.append("Content-Type: text/plain; charset=utf-8")
    lines.append("")  # Ligne vide séparatrice
    
    # Body
    lines.append(email_data.get('body', ''))
    
    return "\n".join(lines)

def extract_identity(from_field):
    """Extrait l'email de l'expéditeur"""
    if isinstance(from_field, str):
        return from_field
    elif isinstance(from_field, dict):
        return from_field.get('email', '')
    else:
        return str(from_field)
```

```python
        # Vérifier les headers GWyl
        for header in msg.keys():

                analysis['delivery_methods_found'].append('headers')

                    header: msg[header]
                }
        
        # Vérifier le corps pour les URLs GWyl
        body = self._extract_body(msg)

            analysis['delivery_methods_found'].append('body_url')
```

### **1. Principe Fondamental**

### **2. Implémentation Technique**
```python
# gwyl_temporal_seal.py
import hashlib
import json
from datetime import datetime
from pathlib import Path

class TemporalEmailSeal:
    def __init__(self):
        self.seal_records = {}
    
    def create_sealed_email(self, email_content, sender_identity):
        """Crée un email scellé avec preuve temporelle"""
        
        # 1. Créer le commit initial (comme Git)
        initial_commit = {
            'content': email_content,
            'author': sender_identity,
            'timestamp': datetime.utcnow().isoformat(),
            'parent_hash': None,  # Premier commit
            'message': 'Initial sealed email'
        }
        
        # 2. Calculer le hash du commit
        commit_hash = self._compute_commit_hash(initial_commit)
        initial_commit['hash'] = commit_hash
        
        # 3. Sceller avec double timestamping
        temporal_proof = self._apply_temporal_seal(commit_hash)
        
        # 4. Stocker l'état scellé
        sealed_state = {
            'initial_commit': initial_commit,
            'temporal_proof': temporal_proof,
            'sealed_at': datetime.utcnow().isoformat(),
            'is_sealed': True
        }
        
        self.seal_records[commit_hash] = sealed_state
        return sealed_state
    
    def open_email(self, sealed_state, recipient_identity):
        """Ouvre l'email et brise le scellé temporel"""
        
        # 1. Vérifier que le scellé est intact
        if not self._verify_seal_intact(sealed_state):
            raise SealBrokenError("Email already opened or tampered with")
        
        # 2. Créer un nouveau commit d'ouverture
        open_commit = {
            'content': sealed_state['initial_commit']['content'],
            'author': recipient_identity,
            'timestamp': datetime.utcnow().isoformat(),
            'parent_hash': sealed_state['initial_commit']['hash'],  # Lien parent
            'message': f'Opened by {recipient_identity}',
            'action': 'open'
        }
        
        # 3. Calculer le nouveau hash (DIFFÉRENT du original)
        open_hash = self._compute_commit_hash(open_commit)
        open_commit['hash'] = open_hash
        
        # 4. Briser le scellé original
        sealed_state['is_sealed'] = False
        sealed_state['open_commit'] = open_commit
        sealed_state['opened_at'] = datetime.utcnow().isoformat()
        sealed_state['opened_by'] = recipient_identity
        
        # 5. Créer une preuve d'ouverture
        open_proof = self._create_open_proof(sealed_state, open_commit)
        
        return {
            'original_seal': sealed_state,
            'open_commit': open_commit,
            'open_proof': open_proof,
            'seal_broken': True
        }
    
    def _compute_commit_hash(self, commit_data):
        """Calcule le hash Git-like du commit"""
        commit_string = json.dumps(commit_data, sort_keys=True, separators=(',', ':'))
        return hashlib.sha256(commit_string.encode()).hexdigest()
    
    def _apply_temporal_seal(self, commit_hash):
        """Applique un scellé temporel cryptographique"""
        # Utiliser Sigstore + OTS comme avant
        from .sigstore_timestamp import sign_and_timestamp
        from .ots_manager import OTSManager
        
        sigstore_proof = sign_and_timestamp(commit_hash.encode(), "temporal_seal")
        ots_proof = OTSManager.submit(commit_hash.encode(), Path("/tmp"))
        
        return {
            'sigstore_seal': sigstore_proof,
            'ots_seal': ots_proof,
            'sealed_hash': commit_hash
        }
    
    def _verify_seal_intact(self, sealed_state):
        """Vérifie que le scellé n'a pas été brisé"""
        if not sealed_state['is_sealed']:
            return False
        
        # Vérifier que le hash original correspond toujours
        current_hash = self._compute_commit_hash(sealed_state['initial_commit'])
        original_hash = sealed_state['initial_commit']['hash']
        
        return current_hash == original_hash

class SealBrokenError(Exception):
    pass
```

### **4. Vérification d'Intégrité Temporelle**
```python
# temporal_verification.py
class TemporalVerifier:
    def verify_email_timeline(self, email_history):
        """Vérifie la timeline complète d'un email scellé"""
        
        timeline_events = []
        
        # 1. Vérifier le scellé initial
        initial_seal = email_history['initial_seal']
        if self._verify_temporal_proof(initial_seal['temporal_proof']):
            timeline_events.append({
                'event': 'sealed',
                'timestamp': initial_seal['sealed_at'],
                'actor': initial_seal['initial_commit']['author'],
                'integrity': 'intact'
            })
        
        # 2. Vérifier les ouvertures successives
        for open_event in email_history.get('open_events', []):
            if self._verify_open_proof(open_event):
                timeline_events.append({
                    'event': 'opened',
                    'timestamp': open_event['opened_at'],
                    'actor': open_event['opened_by'],
                    'integrity': 'seal_broken'
                })
        
        # 3. Vérifier la cohérence de la chaine
        chain_valid = self._verify_commit_chain(email_history)
        
        return {
            'timeline': sorted(timeline_events, key=lambda x: x['timestamp']),
            'chain_integrity': chain_valid,
            'first_sealed_by': initial_seal['initial_commit']['author'],
            'last_opened_by': timeline_events[-1]['actor'] if timeline_events else None
        }
    
    def _verify_commit_chain(self, email_history):
        """Vérifie la chaine Git-like des commits"""
        commits = [email_history['initial_seal']['initial_commit']]
        commits.extend([e['open_commit'] for e in email_history.get('open_events', [])])
        
        for i in range(1, len(commits)):
            current = commits[i]
            parent = commits[i-1]
            
            # Vérifier que le parent_hash pointe vers le bon parent
            if current['parent_hash'] != parent['hash']:
                return False
            
            # Vérifier que le hash est correct
            computed_hash = self._compute_commit_hash(current)
            if computed_hash != current['hash']:
                return False
        
        return True
```

```python
import hashlib
import json
import secrets
from datetime import datetime, timezone
from pathlib import Path
from typing import Dict, List, Optional, Any
import base64

class TemporalSealSystem:
    """
    Système de scellé temporel pour emails basé sur le principe Git
    Chaque email = commit, chaque ouverture = nouveau commit
    """
    
    def __init__(self, storage_path: Path = Path(".gwyl_seals")):
        self.storage_path = storage_path
        self.storage_path.mkdir(exist_ok=True)
        
    def create_sealed_email(self, 
                          from_identity: str,
                          to_identity: str, 
                          subject: str,
                          body: str,
                          attachments: List[Dict] = None) -> Dict[str, Any]:
        """
        Crée un email scellé avec preuve temporelle initiale
        """
        # Générer un ID unique pour cet email
        email_id = f"mail_{secrets.token_hex(8)}_{int(datetime.now(timezone.utc).timestamp())}"
        
        # Construire le commit initial
        initial_commit = {
            'email_id': email_id,
            'type': 'initial_seal',
            'author': from_identity,
            'committer': from_identity,
            'timestamp': datetime.now(timezone.utc).isoformat(),
            'message': f'Initial seal - From: {from_identity}',
            'content': {
                'headers': {
                    'from': from_identity,
                    'to': to_identity,
                    'subject': subject,
                    'date': datetime.now(timezone.utc).isoformat()
                },
                'body': body,
                'attachments': attachments or []
            },
            'parent_hash': None,  # Premier commit de la chaine
            'seal_version': '1.0'
        }
        
        # Calculer le hash du commit
        initial_commit['hash'] = self._compute_commit_hash(initial_commit)
        
        # Appliquer le scellé temporel cryptographique
        temporal_proof = self._apply_temporal_seal(initial_commit['hash'])
        
        # État scellé complet
        sealed_state = {
            'email_id': email_id,
            'initial_commit': initial_commit,
            'temporal_proof': temporal_proof,
            'sealed_at': datetime.now(timezone.utc).isoformat(),
            'sealed_by': from_identity,
            'is_sealed': True,
            'commit_chain': [initial_commit['hash']]  # Début de la chaine
        }
        
        # Sauvegarder l'état scellé
        self._save_sealed_state(sealed_state)
        
        return sealed_state
    
    def open_sealed_email(self, 
                         sealed_state: Dict[str, Any], 
                         opener_identity: str,
                         user_agent: str = "") -> Dict[str, Any]:
        """
        Ouvre un email scellé et crée un commit d'ouverture
        Brise le scellé original et crée une preuve d'accès
        """
        # Vérifier que le scellé est intact
        if not sealed_state['is_sealed']:
            raise SealAlreadyBrokenError(f"Email {sealed_state['email_id']} already opened")
        
        # Vérifier l'intégrité du commit initial
        if not self._verify_commit_integrity(sealed_state['initial_commit']):
            raise SealIntegrityError(f"Initial commit integrity compromised")
        
        # Créer le commit d'ouverture
        open_commit = {
            'email_id': sealed_state['email_id'],
            'type': 'access_open',
            'author': opener_identity,
            'committer': opener_identity,
            'timestamp': datetime.now(timezone.utc).isoformat(),
            'message': f'Opened by {opener_identity} - {user_agent}',
            'content': sealed_state['initial_commit']['content'],  # Même contenu
            'parent_hash': sealed_state['initial_commit']['hash'],  # Lien vers parent
            'access_context': {
                'user_agent': user_agent,
                'opener_identity': opener_identity,
                'original_author': sealed_state['sealed_by']
            }
        }
        
        # Hash du commit d'ouverture (DIFFÉRENT du original)
        open_commit['hash'] = self._compute_commit_hash(open_commit)
        
        # Mettre à jour l'état scellé
        sealed_state['is_sealed'] = False
        sealed_state['opened_at'] = datetime.now(timezone.utc).isoformat()
        sealed_state['opened_by'] = opener_identity
        sealed_state['open_commit'] = open_commit
        sealed_state['commit_chain'].append(open_commit['hash'])
        
        # Créer une preuve d'ouverture
        open_proof = self._create_open_proof(sealed_state)
        
        # Sauvegarder l'état mis à jour
        self._save_sealed_state(sealed_state)
        
        return {
            'sealed_state': sealed_state,
            'open_proof': open_proof,
            'content': sealed_state['initial_commit']['content']
        }
    
    def verify_timeline(self, email_id: str) -> Dict[str, Any]:
        """
        Vérifie la timeline complète d'un email scellé
        """
        sealed_state = self._load_sealed_state(email_id)
        if not sealed_state:
            raise EmailNotFoundError(f"Email {email_id} not found")
        
        timeline = []
        
        # Événement de scellé initial
        timeline.append({
            'type': 'sealed',
            'timestamp': sealed_state['sealed_at'],
            'actor': sealed_state['sealed_by'],
            'commit_hash': sealed_state['initial_commit']['hash'],
            'integrity': self._verify_temporal_proof(sealed_state['temporal_proof'])
        })
        
        # Événements d'ouverture
        if not sealed_state['is_sealed']:
            timeline.append({
                'type': 'opened',
                'timestamp': sealed_state['opened_at'],
                'actor': sealed_state['opened_by'],
                'commit_hash': sealed_state['open_commit']['hash'],
                'integrity': self._verify_commit_integrity(sealed_state['open_commit'])
            })
        
        # Vérifier la chaine de commits
        chain_valid = self._verify_commit_chain(sealed_state)
        
        return {
            'email_id': email_id,
            'timeline': timeline,
            'chain_integrity': chain_valid,
            'current_state': 'opened' if not sealed_state['is_sealed'] else 'sealed',
            'verification_summary': self._generate_verification_summary(timeline, chain_valid)
        }
    
    def _compute_commit_hash(self, commit_data: Dict) -> str:
        """Calcule le hash Git-like d'un commit"""
        # Normaliser le commit pour le hash (sans le hash lui-même)
        commit_copy = commit_data.copy()
        commit_copy.pop('hash', None)
        
        commit_string = json.dumps(commit_copy, sort_keys=True, separators=(',', ':'))
        return hashlib.sha256(commit_string.encode()).hexdigest()
    
    def _apply_temporal_seal(self, commit_hash: str) -> Dict[str, Any]:
        """Applique un scellé temporel cryptographique"""
        try:
            from .sigstore_timestamp import sign_and_timestamp
            from .ots_manager import OTSManager
            
            # Signature immédiate avec Sigstore
            sigstore_proof = sign_and_timestamp(commit_hash.encode(), "temporal_seal_system")
            
            # Ancrage différé avec OTS
            ots_dir = self.storage_path / "ots_proofs"
            ots_proof_file = OTSManager.submit(commit_hash.encode(), ots_dir)
            
            return {
                'sigstore_seal': sigstore_proof,
                'ots_seal': {
                    'proof_file': str(ots_proof_file) if ots_proof_file else None,
                    'submitted_at': datetime.now(timezone.utc).isoformat()
                },
                'sealed_hash': commit_hash,
                'seal_timestamp': datetime.now(timezone.utc).isoformat()
            }
        except Exception as e:
            # Fallback: scellé local seulement
            return {
                'local_seal': {
                    'hash': commit_hash,
                    'timestamp': datetime.now(timezone.utc).isoformat(),
                    'local_signature': hashlib.sha256(f"{commit_hash}{secrets.token_bytes(32)}".encode()).hexdigest()
                },
                'warning': 'external_seal_failed'
            }
    
    def _verify_commit_integrity(self, commit_data: Dict) -> bool:
        """Vérifie l'intégrité d'un commit"""
        computed_hash = self._compute_commit_hash(commit_data)
        return computed_hash == commit_data['hash']
    
    def _verify_commit_chain(self, sealed_state: Dict) -> bool:
        """Vérifie l'intégrité de la chaine de commits"""
        if sealed_state['is_sealed']:
            return True  # Seulement le commit initial
            
        # Vérifier la liaison parent-enfant
        open_commit = sealed_state['open_commit']
        initial_commit = sealed_state['initial_commit']
        
        return (open_commit['parent_hash'] == initial_commit['hash'] and
                self._verify_commit_integrity(initial_commit) and
                self._verify_commit_integrity(open_commit))
    
    def _create_open_proof(self, sealed_state: Dict) -> Dict:
        """Crée une preuve d'ouverture"""
        return {
            'email_id': sealed_state['email_id'],
            'opened_at': sealed_state['opened_at'],
            'opened_by': sealed_state['opened_by'],
            'original_seal_hash': sealed_state['initial_commit']['hash'],
            'open_commit_hash': sealed_state['open_commit']['hash'],
            'proof_timestamp': datetime.now(timezone.utc).isoformat(),
            'chain_integrity': self._verify_commit_chain(sealed_state)
        }
    
    def _save_sealed_state(self, sealed_state: Dict):
        """Sauvegarde l'état scellé"""
        email_id = sealed_state['email_id']
        state_file = self.storage_path / f"{email_id}.json"
        
        with open(state_file, 'w') as f:
            json.dump(sealed_state, f, indent=2)
    
    def _load_sealed_state(self, email_id: str) -> Optional[Dict]:
        """Charge l'état scellé"""
        state_file = self.storage_path / f"{email_id}.json"
        if state_file.exists():
            with open(state_file, 'r') as f:
                return json.load(f)
        return None

class SealAlreadyBrokenError(Exception):
    pass

class SealIntegrityError(Exception):
    pass

class EmailNotFoundError(Exception):
    pass
```

```python
# sealed_email_formatter.py
import email.mime.multipart
import email.mime.text

from email import policy

class SealedEmailFormatter:
    """

                                  from_addr: str,
                                  to_addr: str,
                                  subject: str,
                                  body: str,
                                  attachments: List[Path] = None) -> EmailMessage:

        Crée un message email avec contenu scellé
        """
        # Créer l'état scellé
        sealed_state = self.seal_system.create_sealed_email(
            from_identity=from_addr,
            to_identity=to_addr,
            subject=subject,
            body=body,
            attachments=[{'name': att.name, 'size': att.stat().st_size} for att in (attachments or [])]
        )
        
        # Créer le message email
        msg = EmailMessage()
        msg['From'] = from_addr
        msg['To'] = to_addr
        msg['Subject'] = f"🔐 {subject}"  # Indicateur visuel
        msg['Date'] = email.utils.formatdate(localtime=True)
        msg['X-GWyl-Sealed'] = 'true'
        msg['X-GWyl-Email-ID'] = sealed_state['email_id']
        msg['X-GWyl-Sealed-At'] = sealed_state['sealed_at']
        
        # Corps du message avec instructions
        instructions = self._generate_sealed_instructions(sealed_state)
        body_with_instructions = f"""{body}

{instructions}
"""
        msg.set_content(body_with_instructions)
        
        # Ajouter la preuve scellée en pièce jointe
        proof_attachment = self._create_proof_attachment(sealed_state)
        msg.add_attachment(proof_attachment)
        
        # Ajouter les pièces jointes originales
        for att_path in (attachments or []):
            with open(att_path, 'rb') as f:
                att_data = f.read()
            msg.add_attachment(att_data, 
                             maintype='application', 
                             subtype='octet-stream',
                             filename=att_path.name)
        
        return msg
    
    def _generate_sealed_instructions(self, sealed_state: Dict) -> str:
        """Génère les instructions pour l'email scellé"""
        email_id = sealed_state['email_id']
        verify_url = f"https://verify.gwyl.io/{email_id}"
        
        return f"""
---
🔐 GWYL SEALED EMAIL - TEMPORAL PROOF
Email ID: {email_id}
Sealed at: {sealed_state['sealed_at']}
From: {sealed_state['sealed_by']}

⚠️  IMPORTANT: This email is cryptographically sealed.
   Opening it will break the seal and create a permanent 
   record of access with timestamp and identity.

📋 TO OPEN:
1. Use a GWyl-compatible client
2. Or forward to: open@{email_id}.gwyl.io
3. Or visit: {verify_url}

🔍 VERIFY SEAL: {verify_url}
---
"""
    
    def _create_proof_attachment(self, sealed_state: Dict) -> email.mime.application.MIMEApplication:
        """Crée la pièce jointe avec la preuve scellée"""
        proof_data = {
            'gwyl_version': 'sealed-1.0',
            'email_id': sealed_state['email_id'],
            'sealed_state': sealed_state,
            'verification_info': {
                'verify_url': f"https://verify.gwyl.io/{sealed_state['email_id']}",
                'open_url': f"https://open.gwyl.io/{sealed_state['email_id']}",
                'api_endpoint': f"https://api.gwyl.io/v1/emails/{sealed_state['email_id']}"
            }
        }
        
        proof_json = json.dumps(proof_data, indent=2)
        attachment = email.mime.application.MIMEApplication(
            proof_json,
            _subtype='json',
            name=f"gwyl-proof-{sealed_state['email_id']}.json"
        )
        attachment.add_header(
            'Content-Disposition',
            'attachment',
            filename=f"gwyl-proof-{sealed_state['email_id']}.json"
        )
        
        return attachment
    
    def parse_received_sealed_email(self, email_message: EmailMessage) -> Dict:
        """
        Parse un email reçu pour détecter et extraire le contenu scellé
        """
        # Vérifier si c'est un email scellé GWyl
        if email_message.get('X-GWyl-Sealed') != 'true':
            return {'is_sealed': False, 'content': self._extract_standard_content(email_message)}
        
        email_id = email_message.get('X-GWyl-Email-ID')
        if not email_id:
            return {'is_sealed': False, 'error': 'Missing email ID'}
        
        # Extraire la preuve scellée
        sealed_state = self._extract_sealed_state(email_message)
        if not sealed_state:
            return {'is_sealed': True, 'error': 'Cannot extract sealed state', 'email_id': email_id}
        
        return {
            'is_sealed': True,
            'email_id': email_id,
            'sealed_state': sealed_state,
            'original_headers': {
                'from': email_message['From'],
                'to': email_message['To'],
                'subject': email_message['Subject'],
                'date': email_message['Date']
            }
        }
    
    def _extract_sealed_state(self, email_message: EmailMessage) -> Optional[Dict]:
        """Extrait l'état scellé des pièces jointes"""
        for part in email_message.walk():
            if (part.get_content_type() == 'application/json' and
                'gwyl-proof' in part.get_filename() or ''):
                
                proof_data = json.loads(part.get_payload(decode=True))
                return proof_data.get('sealed_state')
        
        return None
```

```python
# gwyl_sealed_client.py
import smtplib
import getpass

class GWylSealedClient:
    """
    Client email avec support natif des emails scellés
    """
    
    def __init__(self, smtp_server: str = "localhost", smtp_port: int = 587):
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port
        self.seal_system = TemporalSealSystem()
        self.formatter = SealedEmailFormatter(self.seal_system)
        
    def send_sealed_email(self,
                         from_addr: str,
                         to_addr: str,
                         subject: str,
                         body: str,
                         attachments: List[Path] = None,
                         smtp_username: str = None,
                         smtp_password: str = None) -> Dict:
        """
        Envoie un email scellé
        """
        # Créer le message scellé
        sealed_msg = self.formatter.create_sealed_email_message(
            from_addr, to_addr, subject, body, attachments
        )
        
        # Envoyer via SMTP
        with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
            server.starttls()
            
            if smtp_username and smtp_password:
                server.login(smtp_username, smtp_password)
            elif smtp_username:
                password = getpass.getpass(f"Password for {smtp_username}: ")
                server.login(smtp_username, password)
            
            server.send_message(sealed_msg)
        
        # Retourner les métadonnées de preuve
        email_id = sealed_msg['X-GWyl-Email-ID']
        return {
            'status': 'sent',
            'email_id': email_id,
            'sealed_at': sealed_msg['X-GWyl-Sealed-At'],
            'verify_url': f"https://verify.gwyl.io/{email_id}"
        }
    
    def receive_sealed_email(self, email_message: EmailMessage, opener_identity: str) -> Dict:
        """
        Reçoit et ouvre un email scellé
        """
        # Parser l'email
        parsed = self.formatter.parse_received_sealed_email(email_message)
        
        if not parsed['is_sealed']:
            return {
                'is_sealed': False,
                'content': parsed.get('content', ''),
                'opened_at': datetime.now(timezone.utc).isoformat()
            }
        
        if 'error' in parsed:
            return {
                'is_sealed': True,
                'error': parsed['error'],
                'email_id': parsed.get('email_id')
            }
        
        # Ouvrir l'email scellé
        try:
            opened_result = self.seal_system.open_sealed_email(
                parsed['sealed_state'],
                opener_identity,
                user_agent="GWylSealedClient/1.0"
            )
            
            return {
                'is_sealed': True,
                'was_opened': True,
                'email_id': parsed['email_id'],
                'content': opened_result['content'],
                'open_proof': opened_result['open_proof'],
                'original_author': parsed['sealed_state']['sealed_by'],
                'opened_at': opened_result['sealed_state']['opened_at']
            }
            
        except SealAlreadyBrokenError:
            # L'email a déjà été ouvert
            timeline = self.seal_system.verify_timeline(parsed['email_id'])
            return {
                'is_sealed': True,
                'was_opened': False,  # Déjà ouvert avant
                'email_id': parsed['email_id'],
                'content': parsed['sealed_state']['initial_commit']['content'],
                'already_opened_at': parsed['sealed_state'].get('opened_at'),
                'opened_by': parsed['sealed_state'].get('opened_by'),
                'timeline': timeline
            }
    
    def verify_email_timeline(self, email_id: str) -> Dict:
        """
        Vérifie la timeline d'un email scellé
        """
        return self.seal_system.verify_timeline(email_id)
```

```python
# gwyl_simple_seal.py
import hashlib
import json
from datetime import datetime, timezone
from typing import Dict

class SimpleTemporalSeal:
    """
    Système ultra-simple : hash = f(content + author + timestamp)
    Chaque ouverture = nouveau timestamp = nouveau hash
    """
    
    def create_initial_commit(self, from_email: str, to_email: str, subject: str, body: str) -> Dict:
        """Crée le commit initial (email scellé)"""
        commit = {
            'content': {
                'from': from_email,
                'to': to_email, 
                'subject': subject,
                'body': body
            },
            'author': from_email,
            'timestamp': datetime.now(timezone.utc).isoformat(),  # 👈 CLÉ
            'action': 'sealed',
            'parent': None
        }
        
        commit['hash'] = self._compute_hash(commit)
        return commit
    
    def create_open_commit(self, original_commit: Dict, opener: str) -> Dict:
        """Crée un commit d'ouverture (scellé brisé)"""
        open_commit = {
            'content': original_commit['content'],  # Même contenu
            'author': opener,
            'timestamp': datetime.now(timezone.utc).isoformat(),  # 👈 NOUVEAU TIMESTAMP
            'action': 'opened', 
            'parent': original_commit['hash']  # Lien vers parent
        }
        
        open_commit['hash'] = self._compute_hash(open_commit)
        return open_commit
    
    def _compute_hash(self, commit: Dict) -> str:
        """Hash ultra-simple : JSON canonique du commit"""
        # Exclure le hash lui-même du calcul
        commit_copy = commit.copy()
        commit_copy.pop('hash', None)
        
        # Sérialisation canonique
        canonical = json.dumps(commit_copy, sort_keys=True, separators=(',', ':'))
        return hashlib.sha256(canonical.encode()).hexdigest()
    
    def verify_chain(self, commits: list) -> bool:
        """Vérifie la chaine de commits"""
        for i in range(len(commits)):
            current = commits[i]
            
            # Vérifier le hash
            computed_hash = self._compute_hash(current)
            if computed_hash != current['hash']:
                return False
            
            # Vérifier le parent (sauf premier)
            if i > 0 and current['parent'] != commits[i-1]['hash']:
                return False
        
        return True

# Exemple d'usage
seal = SimpleTemporalSeal()

# 1. Alice scelle un email
email_commit = seal.create_initial_commit(
    "alice@company.com", 
    "bob@partner.com",
    "Contract", 
    "Please sign this contract"
)

print(f"🔐 Email scellé: {email_commit['hash']}")

# 2. Bob ouvre l'email (scellé brisé)
open_commit = seal.create_open_commit(email_commit, "bob@partner.com")

print(f"📨 Email ouvert: {open_commit['hash']}")  # 👈 DIFFÉRENT !
print(f"Parent: {open_commit['parent']}")        # 👈 Pointe vers le scellé original

# 3. Vérification
chain_valid = seal.verify_chain([email_commit, open_commit])
print(f"✅ Chaine valide: {chain_valid}")
```

```python
# gwyl_email_tracker.py
class EmailTracker:
    def __init__(self):
        self.seal_system = SimpleTemporalSeal()
        self.email_commits = {}  # email_id -> liste de commits
    
    def track_email_sent(self, from_email: str, to_email: str, subject: str, body: str) -> str:
        """Track un email envoyé"""
        commit = self.seal_system.create_initial_commit(from_email, to_email, subject, body)
        email_id = commit['hash'][:16]  # ID court pour référence
        
        self.email_commits[email_id] = [commit]
        return email_id
    
    def track_email_opened(self, email_id: str, opener: str) -> str:
        """Track l'ouverture d'un email"""
        if email_id not in self.email_commits:
            raise ValueError("Email non trouvé")
        
        original_commit = self.email_commits[email_id][0]
        open_commit = self.seal_system.create_open_commit(original_commit, opener)
        
        self.email_commits[email_id].append(open_commit)
        return open_commit['hash']
    
    def get_email_timeline(self, email_id: str) -> Dict:
        """Retourne la timeline complète d'un email"""
        if email_id not in self.email_commits:
            return {'error': 'Email non trouvé'}
        
        commits = self.email_commits[email_id]
        chain_valid = self.seal_system.verify_chain(commits)
        
        timeline = []
        for commit in commits:
            timeline.append({
                'action': commit['action'],
                'actor': commit['author'], 
                'timestamp': commit['timestamp'],
                'hash': commit['hash']
            })
        
        return {
            'email_id': email_id,
            'timeline': timeline,
            'chain_valid': chain_valid,
            'current_state': 'opened' if len(commits) > 1 else 'sealed'
        }
```
